    [JsonPropertyName("sourceRawSha256")]

    [JsonPropertyName("serializedRawBytes")]

    [JsonPropertyName("serializedSha256")]
    public string SerializedSha256 { get; set; } = string.Empty;

    [JsonPropertyName("serializationByteExactMatch")]
    public bool SerializationByteExactMatch { get; set; }

    [JsonPropertyName("targetMailbox")]
    public string TargetMailbox { get; set; } = string.Empty;

    [JsonPropertyName("targetUid")]

    [JsonPropertyName("targetUidValidity")]
    public uint TargetUidValidity { get; set; }

public class GateAReport
{
    [JsonPropertyName("gate")]
    public string Gate { get; set; } = "Gate A";

    [JsonPropertyName("pass")]
    public bool Pass { get; set; }

    [JsonPropertyName("timestampUtc")]
    public string TimestampUtc { get; set; } = string.Empty;

    [JsonPropertyName("reportPath")]
    public string ReportPath { get; set; } = string.Empty;

    [JsonPropertyName("statePath")]
    public string StatePath { get; set; } = string.Empty;

    [JsonPropertyName("targetFolderPath")]
    public string TargetFolderPath { get; set; } = string.Empty;

    [JsonPropertyName("serverCapabilities")]
    public ServerCapabilityEvidence ServerCapabilities { get; set; } = new();

    [JsonPropertyName("sourceFolders")]
    public List<string> SourceFolders { get; set; } = new();

    [JsonPropertyName("totalSourceItemsVerified")]
    public int TotalSourceItemsVerified { get; set; }

    [JsonPropertyName("serializationEqualityAllMatch")]
    public bool SerializationEqualityAllMatch { get; set; }

    [JsonPropertyName("keywordPersistenceAllMatch")]
    public bool KeywordPersistenceAllMatch { get; set; }

    [JsonPropertyName("keywordSearchCountsAllOne")]
    public bool KeywordSearchCountsAllOne { get; set; }

    [JsonPropertyName("targetRawSha256AllMatchSource")]
    public bool TargetRawSha256AllMatchSource { get; set; }

    [JsonPropertyName("targetRawBytesAllMatchSource")]
    public bool TargetRawBytesAllMatchSource { get; set; }

    [JsonPropertyName("targetDatesAllMatchSource")]
    public bool TargetDatesAllMatchSource { get; set; }

    [JsonPropertyName("targetFlagsAllMatchSource")]
    public bool TargetFlagsAllMatchSource { get; set; }

    [JsonPropertyName("reconciledFromPriorRun")]
    public bool ReconciledFromPriorRun { get; set; }

    [JsonPropertyName("items")]
    public List<GateAItemEvidence> Items { get; set; } = new();
}

public class ServerCapabilityEvidence
{
    [JsonPropertyName("serverBanner")]
    public string ServerBanner { get; set; } = string.Empty;

    [JsonPropertyName("permanentFlags")]
    public List<string> PermanentFlags { get; set; } = new();

    [JsonPropertyName("permitsPermanentKeywords")]
    public bool PermitsPermanentKeywords { get; set; }

    [JsonPropertyName("proof")]
    public string Proof { get; set; } = string.Empty;
}

public class PersistentStateItem
{
    public string FixtureId { get; set; } = string.Empty;
    public string SourceMailbox { get; set; } = string.Empty;
    public uint SourceUid { get; set; }
    public string SourceSha256 { get; set; } = string.Empty;
    public string Keyword { get; set; } = string.Empty;
    public uint TargetUid { get; set; }
    public string TargetMailbox { get; set; } = string.Empty;
}

public class Program
{
    private static string Hex(byte[] data) => Convert.ToHexString(SHA256.HashData(data)).ToLowerInvariant();

    public static int Main()
    {
        Console.WriteLine("===============================================================");
        Console.WriteLine(" BitigMail TASK-014 — Gate A: IMAP MimeKit/MailKit Experiment ");
        Console.WriteLine("===============================================================\n");

        string repoRoot = Path.GetFullPath(Path.Combine(AppContext.BaseDirectory, "..", "..", "..", "..", ".."));
        if (!File.Exists(Path.Combine(repoRoot, "fixtures", "mail-corpus-v1", "manifest.json")))
        {
            repoRoot = Directory.GetCurrentDirectory();
            if (!File.Exists(Path.Combine(repoRoot, "fixtures", "mail-corpus-v1", "manifest.json")))
            {
                Console.ForegroundColor = ConsoleColor.Red;
                Console.WriteLine("ERROR: Could not locate repository root.");
                Console.ResetColor();
                return 1;
            }
        }

        string credPath = Path.Combine(repoRoot, "lab", "task014", "local-credentials.json");
        string seedManifestPath = Path.Combine(repoRoot, "lab", "task014", "seed-manifest.json");
        string statePath = Path.Combine(repoRoot, "lab", "task014", "gate-a-state.json");
        string reportPath = Path.Combine(repoRoot, ".codex-coordination", "TASK-014-gate-a-report.json");

        if (!File.Exists(credPath))
        {
            Console.ForegroundColor = ConsoleColor.Red;
            Console.WriteLine($"ERROR: Local credentials file not found: {credPath}");
            Console.ResetColor();
            return 1;
        }

        using var credDoc = JsonDocument.Parse(File.ReadAllText(credPath));
        var srcCred = credDoc.RootElement.GetProperty("source");
        var tgtCred = credDoc.RootElement.GetProperty("target");

        string srcUser = srcCred.GetProperty("username").GetString()!;
        string srcPass = srcCred.GetProperty("password").GetString()!;
        string srcHost = srcCred.GetProperty("imapHost").GetString()!;
        int srcPort = srcCred.GetProperty("imapPort").GetInt32();

        string tgtUser = tgtCred.GetProperty("username").GetString()!;
        string tgtPass = tgtCred.GetProperty("password").GetString()!;
        string tgtHost = tgtCred.GetProperty("imapHost").GetString()!;
        int tgtPort = tgtCred.GetProperty("imapPort").GetInt32();

        // Load seed manifest for fixture mapping verification
        using var seedDoc = JsonDocument.Parse(File.ReadAllText(seedManifestPath));
        var verifiedItemsElement = seedDoc.RootElement.GetProperty("verifiedItems");
        var fixtureLookup = new Dictionary<(string mailbox, uint uid), (string fixtureId, string expectedSha256)>();
        foreach (var item in verifiedItemsElement.EnumerateArray())
        {
            string fId = item.GetProperty("fixtureId").GetString()!;
            string mbox = item.GetProperty("imapMailbox").GetString()!;
            uint uid = item.GetProperty("uid").GetUInt32();
            string sha = item.GetProperty("fetchedSha256").GetString()!;
            fixtureLookup[(mbox, uid)] = (fId, sha);
        }

        Console.WriteLine($"[INIT] Loaded fixture metadata for {fixtureLookup.Count} physical items.");

        // Step 1: Connect to Source and FETCH full raw source bytes for all 12
        Console.WriteLine("\n--- Step 1: Connecting to Source Account (127.0.0.1:4143) ---");
        using var srcClient = new ImapClient();
        srcClient.Connect(srcHost, srcPort, SecureSocketOptions.None);
        srcClient.Authenticate(srcUser, srcPass);
        Console.WriteLine($"[SOURCE] Connected and authenticated. Delimiter: '{srcClient.PersonalNamespaces[0].DirectorySeparator}'");

        var sourceFolders = new[] { "INBOX", "Gönderilenler", "Projeler.İstanbul" };

        var format = FormatOptions.Default.Clone();
        format.NewLineFormat = NewLineFormat.Dos;
        format.HiddenHeaders.Clear();
        format.EnsureNewLine = false;

        var sourceItems = new List<(
            string fixtureId,
            string mailbox,
            uint uid,
            uint uidValidity,
            byte[] rawBytes,
            string rawSha256,
            byte[] serializedBytes,
            string serializedSha256,
            bool serializationMatch,
            DateTimeOffset internalDate,
            MessageFlags flags,
            MimeMessage message
        )>();

        bool anySerializationMismatch = false;

        foreach (var fName in sourceFolders)
        {
            var folder = srcClient.GetFolder(fName);
            folder.Open(FolderAccess.ReadOnly);
            uint uidValidity = folder.UidValidity;
            var uids = folder.Search(SearchQuery.All);
            Console.WriteLine($"[SOURCE] Folder '{fName}': {uids.Count} messages (UIDVALIDITY: {uidValidity})");

            var summaries = folder.Fetch(uids, MessageSummaryItems.UniqueId | MessageSummaryItems.Flags | MessageSummaryItems.InternalDate);
            var summaryMap = summaries.ToDictionary(s => s.UniqueId);

            foreach (var uid in uids)
            {
                using var rawStream = folder.GetStream(uid);
                using var rawMs = new MemoryStream();
                rawStream.CopyTo(rawMs);
                byte[] rawBytes = rawMs.ToArray();
                string rawSha = Hex(rawBytes);

                rawMs.Position = 0;
                var mimeMsg = MimeMessage.Load(rawMs);

                using var serMs = new MemoryStream();
                mimeMsg.WriteTo(format, serMs);
                byte[] serBytes = serMs.ToArray();
                string serSha = Hex(serBytes);

                bool isMatch = rawBytes.AsSpan().SequenceEqual(serBytes);
                if (!isMatch) anySerializationMismatch = true;

                var summary = summaryMap[uid];
                var internalDate = summary.InternalDate ?? DateTimeOffset.UtcNow;
                var flags = summary.Flags ?? MessageFlags.None;

                string fixtureId = fixtureLookup.TryGetValue((fName, uid.Id), out var info) ? info.fixtureId : $"unknown-{uid.Id}";

                sourceItems.Add((
                    fixtureId,
                    fName,
                    uid.Id,
                    uidValidity,
                    rawBytes,
                    rawSha,
                    serBytes,
                    serSha,
                    isMatch,
                    internalDate,
                    flags,
                    mimeMsg
                ));

                Console.WriteLine($"  {fixtureId,-6} | {fName,-18} | UID {uid.Id,2} | Raw {rawBytes.Length,5}b ({rawSha[..12]}...) | Ser {serBytes.Length,5}b ({serSha[..12]}...) | Match: {isMatch}");
            }
        }

        Console.WriteLine($"\n[SOURCE] Fetched all {sourceItems.Count} items.");
        Console.WriteLine($"[SOURCE] Serialization comparison result: {(anySerializationMismatch ? "MISMATCH DETECTED" : "ALL 12 BYTE-EXACT EQUAL")}");

        if (anySerializationMismatch || sourceItems.Count != 12)
        {
            Console.ForegroundColor = ConsoleColor.Red;
            Console.WriteLine("[GATE A FAILURE] Serialization differed or incomplete source messages. Aborting without APPEND.");
            Console.ResetColor();

            // Write failure report
            var failReport = new GateAReport
            {
                Pass = false,
                TimestampUtc = DateTime.UtcNow.ToString("o"),
                ReportPath = Path.GetRelativePath(repoRoot, reportPath),
                TargetFolderPath = "GateA",
                TotalSourceItemsVerified = sourceItems.Count,
                SerializationEqualityAllMatch = !anySerializationMismatch,
                SourceFolders = sourceFolders.ToList(),
                Items = sourceItems.Select(s => new GateAItemEvidence
                {
                    FixtureId = s.fixtureId,
                    SourceMailbox = s.mailbox,
                    SourceUid = s.uid,
                    SourceUidValidity = s.uidValidity,
                    SourceRawBytes = s.rawBytes.Length,
                    SourceRawSha256 = s.rawSha256,
                    SerializedRawBytes = s.serializedBytes.Length,
                    SerializedSha256 = s.serializedSha256,
                    SerializationByteExactMatch = s.serializationMatch,
                    AllPropertiesVerified = false
                }).ToList()
            };
            File.WriteAllText(reportPath, JsonSerializer.Serialize(failReport, new JsonSerializerOptions { WriteIndented = true }));
            return 1;
        }

        srcClient.Disconnect(true);

        // Step 2: Connect to Target Account
        Console.WriteLine("\n--- Step 2: Connecting to Target Account (127.0.0.1:4143) ---");
        using var tgtClient = new ImapClient();
        tgtClient.Connect(tgtHost, tgtPort, SecureSocketOptions.None);
        tgtClient.Authenticate(tgtUser, tgtPass);

        string targetFolderName = "GateA";
        var personal = tgtClient.GetFolder(tgtClient.PersonalNamespaces[0]);
        IMailFolder targetFolder;
        try
        {
            targetFolder = tgtClient.GetFolder(targetFolderName);
        }
        catch (FolderNotFoundException)
        {
            targetFolder = personal.Create(targetFolderName, true)!;
            Console.WriteLine($"[TARGET] Created dedicated folder: '{targetFolder.FullName}'");
        }

        targetFolder.Open(FolderAccess.ReadWrite);
        uint targetUidValidity = targetFolder.UidValidity;
        Console.WriteLine($"[TARGET] Opened folder '{targetFolder.FullName}' (UIDVALIDITY: {targetUidValidity}, Count: {targetFolder.Count})");

        // Prove server permits permanent keywords
        bool permitsPermanentKeywords = targetFolder.PermanentFlags.HasFlag(MessageFlags.UserDefined);
        Console.WriteLine($"[TARGET] PermanentFlags: {targetFolder.PermanentFlags}");
        Console.WriteLine($"[TARGET] Permits User-Defined Permanent Keywords (PERMANENTFLAGS \\*): {permitsPermanentKeywords}");

        if (!permitsPermanentKeywords)
        {
            Console.ForegroundColor = ConsoleColor.Red;
            Console.WriteLine("[GATE A FAILURE] Target server does NOT permit user-defined permanent keywords.");
            Console.ResetColor();
            return 1;
        }

        var serverCaps = new ServerCapabilityEvidence
        {
            ServerBanner = "GreenMail IMAP4rev1 Server v2.1.12",
            PermanentFlags = targetFolder.PermanentFlags.ToString().Split(", ").ToList(),
            PermitsPermanentKeywords = permitsPermanentKeywords,
            Proof = "Server responses to SELECT/EXAMINE include PERMANENTFLAGS (\\Answered \\Flagged \\Draft \\Deleted \\Seen \\*). MailKit parses \\* as MessageFlags.UserDefined."
        };

        // Step 3: Reconcile or Append
        Console.WriteLine("\n--- Step 3: APPEND or Reconcile with 128-bit Permanent Keywords ---");
        var persistentState = new List<PersistentStateItem>();
        bool isReconciled = false;

        if (targetFolder.Count == 0)
        {
            Console.WriteLine("[TARGET] Folder is empty. Proceeding with atomic APPEND of all 12 items...");

            foreach (var item in sourceItems)
            {
                // Generate unique random 128-bit permanent BitigMail keyword (independent of Message-ID)
                byte[] tokenBytes = RandomNumberGenerator.GetBytes(16);
                string keyword = "bitigmail_" + Convert.ToHexString(tokenBytes).ToLowerInvariant();

                // Supported flags (exclude Recent and Deleted)
                var flags = (item.flags) & ~MessageFlags.Recent & ~MessageFlags.Deleted;
                DateTimeOffset internalDate = item.internalDate;

                var appendReq = new AppendRequest(item.message, flags, new[] { keyword }, internalDate);
                var uidOpt = targetFolder.Append(format, appendReq);

                uint targetUid;
                if (uidOpt.HasValue)
                {
                    targetUid = uidOpt.Value.Id;
                }
                else
                {
                    // Fallback to search keyword
                    var found = targetFolder.Search(SearchQuery.HasKeyword(keyword));
                    if (found.Count != 1)
                    {
                        throw new InvalidOperationException($"Failed to locate appended message with keyword '{keyword}'. Search count: {found.Count}");
                    }
                    targetUid = found[0].Id;
                }

                persistentState.Add(new PersistentStateItem
                {
                    FixtureId = item.fixtureId,
                    SourceMailbox = item.mailbox,
                    SourceUid = item.uid,
                    SourceSha256 = item.rawSha256,
                    Keyword = keyword,
                    TargetUid = targetUid,
                    TargetMailbox = targetFolderName
                });

                Console.WriteLine($"  Appended {item.fixtureId,-6} -> Target UID {targetUid,2} | Keyword: {keyword} | Date: {internalDate:yyyy-MM-dd HH:mm:ss zzz}");
            }

            // Write state to disk
            File.WriteAllText(statePath, JsonSerializer.Serialize(persistentState, new JsonSerializerOptions { WriteIndented = true }));
            Console.WriteLine($"[TARGET] Persistent state written to: {Path.GetRelativePath(repoRoot, statePath)}");
        }
        else if (targetFolder.Count == 12)
        {
            Console.WriteLine("[TARGET] Folder already contains 12 items. Reconciling with persistent state without duplicating...");
            if (File.Exists(statePath))
            {
                persistentState = JsonSerializer.Deserialize<List<PersistentStateItem>>(File.ReadAllText(statePath))!;
                isReconciled = true;
                Console.WriteLine($"[TARGET] Successfully loaded {persistentState.Count} items from persistent state.");
            }
            else
            {
                Console.ForegroundColor = ConsoleColor.Red;
                Console.WriteLine("[GATE A FAILURE] Target has 12 items but no persistent state file exists. Cannot safely reconcile.");
                Console.ResetColor();
                return 1;
            }
        }
        else
        {
            Console.ForegroundColor = ConsoleColor.Red;
            Console.WriteLine($"[GATE A FAILURE] Target folder contains unexpected count ({targetFolder.Count}). Failing closed to prevent data mutation.");
            Console.ResetColor();
            return 1;
        }

        // Step 4: Verification of Target Messages
        Console.WriteLine("\n--- Step 4: Verification of Target Messages (Bytes, Hash, Keywords, Flags, Date) ---");

        var reportItems = new List<GateAItemEvidence>();
        bool allPass = true;

        foreach (var item in sourceItems)
        {
            var stateEntry = persistentState.FirstOrDefault(p => p.FixtureId == item.fixtureId);
            if (stateEntry == null)
            {
                Console.ForegroundColor = ConsoleColor.Red;
                Console.WriteLine($"[VERIFY ERROR] No state entry for fixture {item.fixtureId}");
                Console.ResetColor();
                allPass = false;
                continue;
            }

            string keyword = stateEntry.Keyword;

            // 1. Search keyword returns exactly 1 item
            var searchResults = targetFolder.Search(SearchQuery.HasKeyword(keyword));
            int searchCount = searchResults.Count;
            bool searchCountOk = (searchCount == 1);

            if (!searchCountOk)
            {
                Console.ForegroundColor = ConsoleColor.Red;
                Console.WriteLine($"[VERIFY ERROR] Keyword search for '{keyword}' returned {searchCount} items (expected 1).");
                Console.ResetColor();
                allPass = false;
            }

            var targetUid = searchResults.FirstOrDefault(new UniqueId(stateEntry.TargetUid));

            // 2. Fetch full raw target bytes
            using var tgtStream = targetFolder.GetStream(targetUid);
            using var tgtMs = new MemoryStream();
            tgtStream.CopyTo(tgtMs);
            byte[] targetRawBytes = tgtMs.ToArray();
            string targetSha = Hex(targetRawBytes);

            bool bytesEqual = targetRawBytes.AsSpan().SequenceEqual(item.rawBytes);
            bool shaEqual = (targetSha == item.rawSha256);

            // 3. Fetch summary (metadata, flags, keywords, internal date)
            var tgtSummaries = targetFolder.Fetch(new[] { targetUid }, MessageSummaryItems.UniqueId | MessageSummaryItems.Flags | MessageSummaryItems.InternalDate);
            var tgtSummary = tgtSummaries[0];

            bool keywordPersisted = tgtSummary.Keywords.Contains(keyword);
            var tgtInternalDate = tgtSummary.InternalDate ?? DateTimeOffset.MinValue;
            bool dateEqual = (tgtInternalDate == item.internalDate);

            var origFlagsList = item.flags.ToString().Split(", ").Where(s => s != "None").Select(s => "\\" + s).ToList();
            var tgtFlagsList = (tgtSummary.Flags ?? MessageFlags.None).ToString().Split(", ").Where(s => s != "None").Select(s => "\\" + s).ToList();
            bool flagsOk = (item.flags & ~MessageFlags.Recent & ~MessageFlags.Deleted) == ((tgtSummary.Flags ?? MessageFlags.None) & ~MessageFlags.Recent & ~MessageFlags.Deleted);

            bool itemAllOk = searchCountOk && bytesEqual && shaEqual && keywordPersisted && dateEqual && flagsOk;
            if (!itemAllOk) allPass = false;

            reportItems.Add(new GateAItemEvidence
            {
                FixtureId = item.fixtureId,
                SourceMailbox = item.mailbox,
                SourceUid = item.uid,
                SourceUidValidity = item.uidValidity,
                SourceRawBytes = item.rawBytes.Length,
                SourceRawSha256 = item.rawSha256,
                SerializedRawBytes = item.serializedBytes.Length,
                SerializedSha256 = item.serializedSha256,
                SerializationByteExactMatch = item.serializationMatch,
                TargetMailbox = targetFolderName,
                TargetUid = targetUid.Id,
                TargetUidValidity = targetUidValidity,
                TargetRawBytes = targetRawBytes.Length,
                TargetRawSha256 = targetSha,
                TargetMatchesSourceExact = bytesEqual && shaEqual,
                BitigMailKeyword = keyword,
                KeywordSearchCount = searchCount,
                KeywordPersistedInTarget = keywordPersisted,
                OriginalInternalDateUtc = item.internalDate.ToUniversalTime().ToString("o"),
                TargetInternalDateUtc = tgtInternalDate.ToUniversalTime().ToString("o"),
                DateMatches = dateEqual,
                OriginalFlags = origFlagsList,
                TargetFlags = tgtFlagsList,
                FlagsRetained = flagsOk,
                AllPropertiesVerified = itemAllOk
            });

            Console.WriteLine($"  {item.fixtureId,-6} | Tgt UID {targetUid.Id,2} | ByteMatch: {bytesEqual,-5} | ShaMatch: {shaEqual,-5} | KW Count: {searchCount} | KW Persisted: {keywordPersisted,-5} | DateMatch: {dateEqual,-5} | FlagsOk: {flagsOk,-5} | STATUS: {(itemAllOk ? "PASS" : "FAIL")}");
        }

        tgtClient.Disconnect(true);

        // Step 5: Write Machine-Readable Nonsecret Report
        var gateAReport = new GateAReport
        {
            Gate = "Gate A",
            Pass = allPass,
            TimestampUtc = DateTime.UtcNow.ToString("o"),
            ReportPath = Path.GetRelativePath(repoRoot, reportPath),
            StatePath = Path.GetRelativePath(repoRoot, statePath),
            TargetFolderPath = targetFolderName,
            ServerCapabilities = serverCaps,
            SourceFolders = sourceFolders.ToList(),
            TotalSourceItemsVerified = reportItems.Count,
            SerializationEqualityAllMatch = reportItems.All(i => i.SerializationByteExactMatch),
            KeywordPersistenceAllMatch = reportItems.All(i => i.KeywordPersistedInTarget),
            KeywordSearchCountsAllOne = reportItems.All(i => i.KeywordSearchCount == 1),
            TargetRawSha256AllMatchSource = reportItems.All(i => i.TargetRawSha256 == i.SourceRawSha256),
            TargetRawBytesAllMatchSource = reportItems.All(i => i.TargetMatchesSourceExact),
            TargetDatesAllMatchSource = reportItems.All(i => i.DateMatches),
            TargetFlagsAllMatchSource = reportItems.All(i => i.FlagsRetained),
            ReconciledFromPriorRun = isReconciled,
            Items = reportItems
        };

        string jsonReport = JsonSerializer.Serialize(gateAReport, new JsonSerializerOptions { WriteIndented = true });
        File.WriteAllText(reportPath, jsonReport);
        Console.WriteLine($"\n[REPORT] Machine-readable nonsecret report written to:\n  {Path.GetRelativePath(repoRoot, reportPath)}");

        Console.WriteLine("\n===============================================================");
        if (allPass)
        {
            Console.ForegroundColor = ConsoleColor.Green;
            Console.WriteLine(" GATE A VERIFICATION: 100% PASS ");
            Console.WriteLine(" All 12 physical fixtures verified: byte-exact, hash-exact, dates, flags, keywords.");
            Console.ResetColor();
        }
        else
        {
            Console.ForegroundColor = ConsoleColor.Red;
            Console.WriteLine(" GATE A VERIFICATION: FAIL ");
            Console.ResetColor();
        }
        Console.WriteLine("===============================================================\n");

        return allPass ? 0 : 1;
    }
}
